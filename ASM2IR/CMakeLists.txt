cmake_minimum_required(VERSION 3.16)
project(WaveASM LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION}")
add_executable(asm2ir main.cpp isa.cpp ir.cpp runtime.cpp)
target_include_directories(asm2ir SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
if(TARGET LLVM)
    target_link_libraries(asm2ir PRIVATE LLVM)
else()
    llvm_map_components_to_libnames(LLVM_LIBS core executionengine mcjit native)
    target_link_libraries(asm2ir PRIVATE ${LLVM_LIBS})
endif()
target_compile_options(asm2ir PRIVATE -Wall -Wextra -Wpedantic)
find_package(SDL2 QUIET)
if(SDL2_FOUND)
    target_compile_definitions(asm2ir PRIVATE WAVE_SDL)
    target_include_directories(asm2ir PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(asm2ir PRIVATE ${SDL2_LIBRARIES})
else()
    message(STATUS "SDL2 development files not found: building headless runner")
endif()
add_executable(wave_reference tests/reference.cpp ../app.c runtime.cpp)
target_include_directories(wave_reference PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME wave_asm_check COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.py
    $<TARGET_FILE:asm2ir> $<TARGET_FILE:wave_reference> ${CMAKE_CURRENT_SOURCE_DIR}/wave.s)
set_tests_properties(wave_asm_check PROPERTIES TIMEOUT 300)
